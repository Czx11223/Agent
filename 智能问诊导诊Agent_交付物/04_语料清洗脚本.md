# 问诊对话语料收集与清洗脚本

- 数据来源：`Chinese-Dialogue-Dataset`（[GitHub 索引](https://github.com/YouTaoBaBa/Chinese-Dialogue-Dataset)）指向的[阿里天池中文医疗对话数据集](https://tianchi.aliyun.com/dataset/90163)（约 79.2 万条问答对）。
- 目标：筛出**医疗问诊类**对话，过滤诊断/开药类内容，把医患问答拆分为样本结构【患者主诉】→【医生追问】→【患者补充回答】，输出标准化 CSV。
- 语言/运行环境：Python 3.12+，仅依赖标准库 `json / re / csv / argparse / pathlib / collections`，开箱即用。

---

## 1. 总体处理流程

```
原始 JSON（医患问答集）
   │  ① 加载与编码识别（utf-8 / gbk / utf-8-sig）
   ▼
  ② 结构归一：schema 自动探测 → 统一为咨询记录(科室/症状/病情描述/轮次)
   │
   ▼
  ③ 对话级过滤：仅保留医疗问诊类（医学科室白名单 + 医疗语义关键词）
   │
   ▼
  ④ 轮次拆分与分类：患者主诉 / 医生追问 / 患者补充；医生回答按内容重分类
   │
   ▼
  ⑤ 轮次级清洗：文本规范化、去脏数据（链接/emoji/异常字符）并做质量评分
   │
   ▼
  ⑥ 黑名单过滤：命中诊断/处方/用药剂量/检查治疗医嘱 → 剔除该轮次或整个样本
   │
   ▼
  ⑦ 三元组构建：从每段医患问答生成【主诉|追问|补充】训练样本
   │
   ▼
  ⑧ 去重 → 洗牌 → 写出 CSV + 清洗统计报告
```

## 2. 完整清洗脚本代码 `clean_inquiry_corpus.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
问诊对话语料收集与清洗脚本
数据来源: Chinese-Dialogue-Dataset -> 阿里天池中文医疗对话数据集(约79.2万问答对)
目标    : 过滤医疗问诊类对话,拆分轮次,过滤诊断/开药内容,输出【主诉|医生追问|患者补充】标准化CSV
依赖    : 仅标准库 (python3.12+)
"""

from __future__ import annotations

import argparse
import csv
import json
import re
import sys
from collections import Counter
from pathlib import Path

# ============================================================================
# 一、可配置规则
# ============================================================================

# 医学/临床科室白名单：用于判定对话是否为"医疗问诊类"（非问诊则整条过滤）
MEDICAL_DEPARTMENTS = [
    "内科", "外科", "呼吸", "消化", "心内", "心外", "神经", "内分泌",
    "耳鼻喉", "眼科", "口腔", "皮肤", "妇科", "产科", "儿科", "泌尿",
    "骨科", "风湿", "肿瘤", "血液", "感染", "急诊", "中医", "康复",
    "精神", "过敏", "睡眠", "胃肠", "肝胆", "甲状腺", "肾内", "免疫",
    "头晕", "疼痛", "发热门诊", "心血管", "呼吸内科", "消化内科",
]

# 医疗语义关键词：被交叉验证为问诊类对话（用于无科室字段时兜底判定）
MEDICAL_SEMANTIC_KEYWORDS = [
    "症状", "疼痛", "发烧", "发热", "咳嗽", "呕吐", "腹泻", "头晕", "头痛",
    "胸闷", "气短", "呼吸困难", "胃痛", "乏力", "皮疹", "失眠", "血压",
    "血糖", "用药", "病史", "多久了", "什么时候", "持续", "加重",
]

# ----------------------------------------------------------------------------
# 黑名单（禁止样本）：命中的样本/轮次将**剔除**
# 核心原则：预问诊只收集症状要素(主诉/部位/时间/伴随/既往), 不做诊断、不开药。
# ----------------------------------------------------------------------------

# A. 诊断类结论正则（给出明确诊断意见 -> 剔除轮次）
PATTERN_DIAGNOSIS = [
    r"诊断(为|结果|考虑)?[\s:：]?[（(]?[一二三四五六七八九十\d]、?[）)]?",  # "诊断为(1)..."
    r"(可能|疑似|初步)?诊断为[\u4e00-\u9fa5]{2,10}(症|炎|癌|瘤|病|综合征)",
    r"(确诊|临床诊断|确诊为)",
    r"属于[\u4e00-\u9fa5]{2,10}(症|炎|癌|瘤|病)",        # "属于细菌性咽炎"
    r"(表现|符合)[\u4e00-\u9fa5]{2,10}(的诊断标准|病理性)",
    r"(大概率|很可能是)[\u4e00-\u9fa5]{2,10}(病|症|炎)",
]

# B. 处方/用药/治疗医嘱类正则（开药、剂量、治疗方案 -> 剔除该轮次）
PATTERN_MEDICATION = [
    r"(建议|可|应该|需|推荐)(服用|口服|外?用|肌注|静脉|输液|使用|按时)?[\d.]+(mg|g|ml|片|粒|支|次)[^。]{0,20}",
    r"(每日|每天|一天)[\d.\-~至]{1,6}(次|片|粒|mg|g)",
    r"(口服|外用|涂以|肌肉注射|静脉滴注|皮下注射)[^。]{0,20}",
    r"(处方|开药|用药建议|剂量|用量|疗程)[:：]?[^。]{0,20}",
    r"[\u4e00-\u9fa5]{2,6}(片|胶囊|颗粒|口服液|注射液|软膏|喷雾剂)\s*[，,。](建议|每次|每日)",
    r"(建议|请)(尽快?|立即)?(手术|住院|化疗|放疗|复查|就诊|检查|查个?)[^。]{0,15}",
    r"(手术|介入|切除|放疗|化疗)(治疗|方案|指针|适应症)",
    r"(使用|服用)(药物|抗生素|消炎药|退烧药|止痛药)[^。]{0,20}",
]

# C. 诊疗动作/检查医嘱（预问诊阶段不应出现的"执行型"指令）
PATTERN_TREATMENT_ACTION = [
    r"(请|建议|需)(马上去|尽快去|去|就近)(急诊|医院|门诊|药店)买?",
    r"(做|查|行)(个|一下)?(血常规|尿常规|CT|核磁|心电图|B超|胃镜|肠镜|X光|彩超)",
    r"(预约|挂)(个?)号",  # 仅挂号的非问诊引流样本，视需求而定（默认可保留）
]

# D. 其他脏文本/隐私/风险模式（整个样本剔除）
PATTERN_DIRTY = [
    r"https?://\S+",          # URL
    r"www\.\S+",              # 网址
    r"\d{11,}",               # 超长数字（疑似手机号）
    r"[ＱＱqQ]{2}\s*[:：]?\s*\d{5,12}",  # QQ号
    r"\d{3}-\d{4}-\d{4}",     # 国内固定电话
    r"18\d{9}|13\d{9}|15\d{9}",  # 常见手机号段(额外保险)
]

# 语义敏感词（命中即整条删除：含真实身份/联系方式类泄露）
SENSITIVE_WORDS = [
    "身份证", "手机号", "银行卡", "住址", "家庭住址", "微信号",
    "电话是", "加我微信", "q q", "qq群",
]

# 非问诊类/闲聊/杂项对话（命中多条则判定非问诊样本）
NON_INQUIRY_KEYWORDS = [
    "笑死", "哈哈哈", "好嘞", "谢谢医生再见", "闲聊", "游戏", "电影",
    "搞笑", "八卦", "减肥餐单", "财经", "房产",
]

# ============================================================================
# 二、schema 自动探测与归一
# ============================================================================

class SchemaMapping:
    """把数据集的不同字段命名统一为内部字段。"""
    def __init__(self, dept=None, symptom=None, desc=None, ask=None, answer=None):
        self.dept = dept
        self.symptom = symptom
        self.desc = desc
        self.ask = ask      # 患者提问(主诉)字段名
        self.answer = answer  # 医生回答字段名

# 常见候选字段名
KNOWN = {
    "dept": ["科室", "dep", "department"],
    "symptom": ["症状", "symptom", "症状描述"],
    "desc": ["病情描述", "desc", "prescription", "主诉", "present", "病史"],
    "ask": ["问", "question", "pat", "p1", "ask", "患者问题", "speak"],
    "answer": ["答", "answer", "doc", "r1", "医生回答", "reply"],
}


def detect_schema(record: dict) -> SchemaMapping:
    """根据首条记录字段名自动探测 schema，找不到则返回 None。"""
    keys = set(record.keys())
    mapping = SchemaMapping()
    for field, cands in KNOWN.items():
        for c in cands:
            if c in keys:
                setattr(mapping, field, c)
                break
    # 至少要有 问/答 之一，否则视为无法解析
    if not mapping.ask and not mapping.answer:
        return None
    return mapping


# ============================================================================
# 三、文本清洗与质量评分
# ============================================================================

RE_EMOJI = re.compile(
    "[" u"\U0001F600-\U0001F64F" u"\U0001F300-\U0001F5FF"
    u"\U0001F680-\U0001F6FF" u"\U0001F1E0-\U0001F1FF"
    u"\U00002702-\U000027B0" u"\U000024C2-\U0001F251" u"\u2600-\u27BF"
    "]", flags=re.UNICODE)

RE_NON_PRINT = re.compile(r"[\x00-\x08\x0b\x0c\x0e-\x1f]")


def clear_text(text: str) -> str:
    """规范化文本：去头尾空白、折叠空白、去 emoji/控制字符/链接。"""
    if not isinstance(text, str):
        return ""
    t = text.replace("\u3000", " ").replace("\t", " ")
    t = RE_NON_PRINT.sub("", t)
    t = RE_EMOJI.sub("", t)
    t = re.sub(r"https?://\S+|www\.\S+", " ", t)
    t = re.sub(r"\s+", " ", t).strip()
    return t


def chinese_ratio(text: str) -> float:
    """中文字符占比（用于过滤乱码/纯符号文本）。"""
    if not text:
        return 0.0
    han = sum(1 for ch in text if "\u4e00" <= ch <= "\u9fff")
    return han / len(text)


def hit_any(text: str, patterns: list) -> bool:
    return any(re.search(p, text) for p in patterns)


def hit_any_word(text: str, words: list) -> bool:
    return any(w in text for w in words)


# ============================================================================
# 四、轮次分类：患者话/医生话，以及医生话中的"追问"与"诊疗结论"判别
# ============================================================================

def classify_patient_turn(idx: int, text: str) -> str:
    """首条患者话记为主诉(chief)，其后第n条患者补充为补充回答(follow)。"""
    return "chief" if idx == 0 else "follow"


def is_doctor_followup(text: str) -> bool:
    """判断医生话是否为'追问症状'（问诊类），而非 诊断/开药/医嘱。"""
    # 命中诊断/开药/治疗动作黑名单 -> 不是追问,剔除
    if hit_any(text, PATTERN_DIAGNOSIS + PATTERN_MEDICATION + PATTERN_TREATMENT_ACTION):
        return False
    # 是疑问句或含追问要素词 -> 追问
    is_question = text.endswith("?") or text.endswith("？") or "吗" in text or "呢" in text
    has_query_kw = any(k in text for k in [
        "什么", "哪里", "哪个", "多久", "多长", "什么时候", "是否", "有没有",
        "伴随", "之前", "原来", "吃过", "用药", "过敏", "既往", "怎么", "为何", "感觉",
    ])
    return is_question or has_query_kw


# ============================================================================
# 五、三元组构建
# ============================================================================

def build_triples(consult: dict) -> list:
    """
    从一条归一化咨询记录生成若干【主诉|医生追问|患者补充回答】三元组。
    consult 字段: {dept, symptom, desc, turns:[(role, text), ...]}  role in (pat, doc)
    """
    triples = []
    # 分离患者话与医生话
    patient_utts = [text for role, text in consult["turns"] if role == "pat" and text]
    doctor_utts = [text for role, text in consult["turns"] if role == "doc" and text]

    # 主诉：优先取第一条患者话；若没有则退避到病情描述 desc
    chief = patient_utts[0] if patient_utts else consult["desc"]
    if not chief:
        return []

    # 重建带 role 的轮次，便于按"追问->补充"对齐
    sequence = [(role, _) for role, _ in consult["turns"] if _]
    for i, (role, text) in enumerate(sequence):
        if role != "doc":
            continue
        if not is_doctor_followup(text):
            continue  # 诊断/开药/医嘱类医生话剔除
        # 找该医生追问之后下一句患者补充
        follow = next((t for r, t in sequence[i + 1:] if r == "pat" and t), "")
        if not follow:
            continue
        triples.append({
            "chief_complaint": chief,
            "doc_followup": text,
            "patient_answer": follow,
        })
    return triples


# ============================================================================
# 六、顶层清洗流程
# ============================================================================

def is_medical_inquiry(dept: str, symptom: str, desc: str, seq_texts: list) -> bool:
    """对话级过滤：是否为'医疗问诊类'对话。"""
    # 1) 科室白名单命中 -> 是
    if dept and hit_any_word(dept, MEDICAL_DEPARTMENTS):
        return True
    # 2) 语义关键词命中(≥2) -> 是
    hay = "".join(seq_texts[:6]) + (symptom or "") + (desc or "")
    cnt = sum(1 for k in MEDICAL_SEMANTIC_KEYWORDS if k in hay)
    return cnt >= 2


def sample_quality(texts: list) -> float:
    """样本质量分 0~1：长度适中 + 中文占比 + 无脏信息。"""
    joined = "".join(texts)
    n = len(joined)
    if n < 8:
        return 0.0
    ratio = chinese_ratio(joined)
    # 中文占比过低(乱码)直接0分
    if ratio < 0.5:
        return 0.0
    score = min(1.0, n / 120) * 0.5 + min(1.0, ratio) * 0.3 + 0.2
    return round(min(1.0, score), 2)


def clean_one_record(record, mapping, collect_stats) -> list:
    """处理单条原始记录 -> 返回清洗后的三元组列表。"""
    if not isinstance(record, dict):
        return []
    # 提取原始字段
    dept = str(record.get(mapping.dept, "")) if mapping.dept else ""
    symptom = str(record.get(mapping.symptom, "")) if mapping.symptom else ""
    desc = str(record.get(mapping.desc, "")) if mapping.desc else ""
    raw_ask = record.get(mapping.ask, []) if mapping.ask else []
    raw_answer = record.get(mapping.answer, []) if mapping.answer else []
    # 兼容 ask/answer 可能为 list 或 str
    ask_list = raw_ask if isinstance(raw_ask, list) else [raw_ask]
    ans_list = raw_answer if isinstance(raw_answer, list) else [raw_answer]

    # 轮次拆分（交错患者问 / 医生答）
    turns = []
    for a in ask_list:
        c = clear_text(a)
        if c:
            turns.append(("pat", c))
    for r2 in ans_list:
        c = clear_text(r2)
        if c:
            turns.append(("doc", c))
    if not turns:
        collect_stats["empty_turns"] += 1
        return []

    seq_texts = [t for _, t in turns]
    # 全样本敏感/脏检查
    joined = "".join(seq_texts)
    if hit_any_word(joined, SENSITIVE_WORDS) or hit_any(joined, PATTERN_DIRTY):
        collect_stats["dirty_filtered"] += 1
        return []
    if hit_any_word(joined, NON_INQUIRY_KEYWORDS):
        collect_stats["non_inquiry_filtered"] += 1
        return []

    # 对话级：医疗问诊类过滤
    if not is_medical_inquiry(dept, symptom, desc, seq_texts):
        collect_stats["non_inquiry_filtered"] += 1
        return []

    consult = {"dept": dept, "symptom": symptom, "desc": desc, "turns": turns}
    triples = build_triples(consult)
    if not triples:
        collect_stats["no_followup"] += 1
        return []

    # 三元组级过滤与质量判定
    out = []
    for tp in triples:
        parts = [tp["chief_complaint"], tp["doc_followup"], tp["patient_answer"]]
        score = sample_quality(parts)
        if score < 0.6:  # 低于阈值的低质样本丢弃
            collect_stats["low_quality"] += 1
            continue
        tp["quality_score"] = score
        out.append(tp)
    return out


# ============================================================================
# 七、入口
# ============================================================================

def main():
    ap = argparse.ArgumentParser(description="问诊对话语料清洗")
    ap.add_argument("--input", "-i", nargs="+", required=True,
                    help="输入 JSON 文件路径（可多个/通配）")
    ap.add_argument("--out", "-o", default="inquiry_corpus_clean.csv",
                    help="输出 CSV 路径")
    ap.add_argument("--report", "-r", default="clean_report.json",
                    help="清洗统计报告路径")
    ap.add_argument("--no-shuffle", action="store_true", help="不随机打乱")
    args = ap.parse_args()

    stats = Counter()
    results = []
    seen = set()
    idx = 0

    for raw in args.input:
        for path in Path().glob(raw) if any(c in raw for c in "*?[") else [Path(raw)]:
            if not path.exists():
                print(f"[skip] 不存在: {path}", file=sys.stderr)
                continue
            data = load_json_with_encodings(path)
            if data is None:
                stats["unreadable"] += 1
                continue
            if not data:
                continue
            if isinstance(data, dict):
                # 兼容 "data"/"conversation" 包裹
                data = data.get("data") or data.get("conversations") or data.get("items") or list(data.values())[:1]
            schema = detect_schema(data[0])
            if schema is None:
                stats["schema_unknown"] += 1
                print(f"[warn] 无法识别 schema: {path.name}", file=sys.stderr)
                continue
            stats["raw_records"] += len(data)
            for rec in data:
                for tp in clean_one_record(rec, schema, stats):
                    dedup_key = (tp["chief_complaint"], tp["doc_followup"])
                    if dedup_key in seen:
                        stats["duplicate"] += 1
                        continue
                    seen.add(dedup_key)
                    idx += 1
                    results.append({
                        "id": f"S{idx:06d}",
                        "chief_complaint": tp["chief_complaint"],
                        "doc_followup": tp["doc_followup"],
                        "patient_answer": tp["patient_answer"],
                        "department": rec.get(schema.dept, ""),
                        "symptom": rec.get(schema.symptom, ""),
                        "source_file": path.name,
                        "quality_score": tp["quality_score"],
                    })

    if not args.no_shuffle:
        random_shuffle(results)

    write_csv(args.out, results)
    stats["kept_samples"] = len(results)
    stats["filter_rate"] = round(
        1 - (stats["kept_samples"] / max(1, stats["raw_records"])), 4)
    Path(args.report).write_text(json.dumps(dict(stats), ensure_ascii=False, indent=2),
                                 encoding="utf-8")
    print(f">> 保留样本: {len(results)}  统计: {json.dumps(dict(stats), ensure_ascii=False)}")


# ============================================================================
# 工具：带编码识别的 JSON 加载、CSV 写出、随机洗牌
# ============================================================================

def load_json_with_encodings(path: Path):
    encodings = ["utf-8", "utf-8-sig", "gbk", "gb18030", "latin-1"]
    for enc in encodings:
        try:
            with open(path, "r", encoding=enc) as f:
                return json.load(f)
        except (UnicodeDecodeError, json.JSONDecodeError):
            continue
        except Exception:  # noqa
            continue
    return None


def write_csv(path: str, rows: list):
    cols = ["id", "chief_complaint", "doc_followup", "patient_answer",
            "department", "symptom", "source_file", "quality_score"]
    with open(path, "w", newline="", encoding="utf-8-sig") as f:
        w = csv.DictWriter(f, fieldnames=cols, quoting=csv.QUOTE_ALL)
        w.writeheader()
        for r in rows:
            w.writerow(r)


def random_shuffle(rows: list):
    """无第三方依赖的洗牌（稳定性足够的简单实现）。"""
    import random
    random.seed(20261008)
    for i in range(len(rows) - 1, 0, -1):
        j = random.randint(0, i)
        rows[i], rows[j] = rows[j], rows[i]


if __name__ == "__main__":
    main()
```

```bash
# 单文件
python clean_inquiry_corpus.py -i ./data/肠胃科.json -o inquiry_clean.csv -r report.json
```

## 4. 输出的标准化 CSV 结构

| 列名 | 类型 | 说明 |
| --- | --- | --- |
| `id` | TEXT | 样本唯一编号 `S000001`… |
| `chief_complaint` | TEXT | 【患者主诉】：首句病情描述/首条患者话 |
| `doc_followup` | TEXT | 【医生追问】：被保留的医生追问（诊断/开药已剔除） |
| `patient_answer` | TEXT | 【患者补充回答】：追问后患者的补充 |
| `department` | TEXT | 科室（原始字段，可为空） |
| `symptom` | TEXT | 症状要素（原始字段，可为空） |
| `source_file` | TEXT | 来源文件（可溯源） |
| `quality_score` | FLOAT | 样本质量分 0~1（≥0.6 才保留） |

## 5. 异常脏数据过滤逻辑（脚本内实现要点）

| 层级 | 判定 | 处理 |
| --- | --- | --- |
| 编码异常 | utf-8 / gbk / gb18030 / latin-1 均无法解析 JSON | 跳过该文件，计入 `unreadable` |
| schema 无法识别 | 记录无"问/答"候选字段 | 跳过该文件，计入 `schema_unknown` |
| 空轮次 | 清洗后无任何有效话语 | 丢弃，计入 `empty_turns` |
| 敏感泄露 | 含身份证/手机号/QQ/地址/明文联系 | 整条丢弃，计入 `dirty_filtered` |
| 非问诊 | 无医学科室命中且医疗关键词<2，或命中间聊词 | 整条丢弃，计入 `non_inquiry_filtered` |
| 诊断/开药 | 医生话命中 `PATTERN_DIAGNOSIS/MEDICATION/TREATMENT_ACTION` | 该医生轮次剔除，不计入追问 |
| 追问缺失 | 找不到医患"追问→补充"配对 | 丢弃该段，计入 `no_followup` |
| 低质 | 质量分 < 0.6（过短/乱码/中文占比低） | 丢弃，计入 `low_quality` |
| 重复 | 主诉+追问组合已出现 | 丢弃，计入 `duplicate` |
